import numpy as np
from tactile_core.calibration.dataset import ProjectData

rng = np.random.default_rng(0)
pd = ProjectData()
for i in range(1500):
    lat = rng.uniform(0, 1, 4)
    v = lat @ rng.normal(size=(4, 25)) * 0.5 + 1.0
    w = [lat[0] - lat[1], lat[2] ** 2, -5 * lat[3], 0.01 * lat[1], 0.0, 0.02 * lat[0]]
    pd.add_sample(float(i), v, w, [0, 0, 0, 0, 0, 0, 1], zone_index=1, fn_index=1)
pd.save("calibration/synthetic_test")
print("saved", pd.num_samples())
